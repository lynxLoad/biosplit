func _ready() -> void:
	if character_stats == null:
		push_error("Character: character_stats не задан на %s" % name)
		character_stats = CharacterStats.new()
	character_stats = character_stats.duplicate()

	_health = HealthComponent.new()
	_health.setup(character_stats.max_health)

	if not _idle_point_set:
		reset_idle_point()

	if animated_sprite:
		_sprite_base_position = animated_sprite.position
		_setup_services()
		_anim.play(ANIM_IDLE)

# ── state queries ──

	return _state != State.IDLE and _state != State.DEAD


func is_dead() -> bool:
	return _state == State.DEAD


func is_attacking() -> bool:
	return _state == State.ATTACKING or _state == State.RETREATING


func is_blocking() -> bool:
	return _state == State.BLOCKING

	return _health.health if _health else 0.0

	return _health.max_health if _health else 0.0

	z_index = priority

	if character_stats == null:
		return 0.0
	return maxf(0.0, character_stats.block_damage_reduction)

	return _death.get_corpse_anchor() if _death else global_position


# ── public actions ──

func attack(enemy: Character) -> void:
	if _state not in [State.IDLE, State.MOVING_TO_IDLE]:
		return
	if not is_instance_valid(enemy) or enemy.is_dead():
		return

	var id := _begin_action()
	_state = State.ATTACKING
	target = enemy
	_z_before_attack = z_index
	z_index = z_attack_priority

	var start := global_position
	var dir := start - enemy.global_position
	if dir.length_squared() < 0.0001:
		dir = Vector2.RIGHT
	dir = dir.normalized()

	var strike_pos := start
	if start.distance_to(enemy.global_position) > strike_distance:
		strike_pos = enemy.global_position + dir * strike_distance

	var speed_px := maxf(character_stats.speed, 0.01) * pixels_per_unit
	var approach_dist := start.distance_to(strike_pos)
	var retreat_dist := strike_pos.distance_to(idle_point)
	var approach_time := clampf(approach_dist / speed_px, min_phase_duration, max_phase_duration)
	var retreat_time := clampf(retreat_dist / speed_px, min_phase_duration, max_phase_duration)

	if approach_dist > 1.0:
		var ok := await _move.move_to(
			strike_pos, approach_time, id, Tween.TRANS_QUAD, Tween.EASE_OUT
		)
		if not ok:
			return

	var pick := _pick_attack()
	var attack_anim: StringName = pick["anim"]
	var impact_frame := int(pick["frame"])

	_anim.play(attack_anim)
	_juice.play_lunge(dir)

	if weapon:
		await _juice.play_anticipation(attack_anim, impact_frame, _ctx, id)
		if not _ctx.is_current(id): return
		await _anim.await_frame(attack_anim, impact_frame, _ctx, id)
		if not _ctx.is_current(id): return
		if is_instance_valid(enemy) and not enemy.is_dead():
			enemy.take_damage(weapon.damage, global_position)

	_juice.play_impact(attack_anim)
	_juice.play_attack_recoil(dir)

	await _anim.await_end(attack_anim, _ctx, id)
	if not _ctx.is_current(id):
		return

	retreat_started.emit()
	_state = State.RETREATING

	var ok := await _move.move_to(
		idle_point, retreat_time, id, Tween.TRANS_QUAD, Tween.EASE_IN
	)
	if not ok:
		return

	z_index = _z_before_attack
	_state = State.IDLE
	_anim.play(ANIM_IDLE)
	turn_finished.emit()


func block() -> void:
	if _state not in [State.IDLE, State.MOVING_TO_IDLE]:
		return
	_begin_action()
	_state = State.BLOCKING
	_block.reset()
	if _anim.has(ANIM_BLOCK):
		_anim.play(ANIM_BLOCK)
	block_started.emit()


func unblock() -> void:
	if _state != State.BLOCKING:
		return
	_block.reset()
	_state = State.IDLE
	if _anim.has(ANIM_IDLE):
		_anim.play(ANIM_IDLE)
	block_ended.emit()


func take_damage(amount: float, from_position: Vector2 = Vector2.INF) -> void:
	if _state == State.DEAD:
		return

	_last_hit_from = from_position

	if _state == State.BLOCKING:
		await _on_blocked_hit(amount, from_position)
		return

	var pick := _pick_damage()
	var damage_anim: StringName = pick["anim"]

	if _state != State.IDLE or not _anim.has(damage_anim):
		_apply_damage(amount)
		return

	var id := _begin_action()
	_state = State.HIT
	_anim.play(damage_anim)

	await _anim.await_frame(damage_anim, int(pick["frame"]), _ctx, id)
	if not _ctx.is_current(id):
		return

	_apply_damage(amount)

	await _anim.await_end(damage_anim, _ctx, id)
	if not _ctx.is_current(id):
		return

	_state = State.IDLE
	if _anim.has(ANIM_IDLE):
		_anim.play(ANIM_IDLE)


func move_to_idle_point(duration: float = 0.3) -> void:
	if _state not in [State.IDLE, State.MOVING_TO_IDLE]:
		return

	var id := _begin_action()
	_state = State.MOVING_TO_IDLE

	var ok := await _move.move_to(
		idle_point, duration, id, Tween.TRANS_QUAD, Tween.EASE_IN_OUT
	)
	if not ok:
		return

	_state = State.IDLE
	_anim.play(ANIM_IDLE)
	turn_finished.emit()


func heal(amount: float) -> void:
	if _state == State.DEAD or amount <= 0.0:
		return
	_health.heal(amount)


# ── hooks for subclasses ──

func _pick_attack() -> Dictionary:
	if _attack_frames.is_empty():
		return {
			"anim": ANIM_ATTACK,
			"frame": weapon.animation_frame_attack if weapon else 1,
		}

	var available: Array[StringName] = []
	var total_weight := 0.0

	for anim in _attack_frames:
		if _anim.has(anim):
			available.append(anim)
			total_weight += _attack_weight(anim)

	if available.is_empty():
		return {
			"anim": ANIM_ATTACK,
			"frame": weapon.animation_frame_attack if weapon else 1,
		}

	if total_weight <= 0.0:
		var fallback: StringName = available.pick_random()
		return {"anim": fallback, "frame": _attack_frames[fallback]}

	var roll := randf() * total_weight
	var chosen: StringName = available[available.size() - 1]

	for anim in available:
		roll -= _attack_weight(anim)
		if roll <= 0.0:
			chosen = anim
			break

	return {"anim": chosen, "frame": _attack_frames[chosen]}


func _attack_weight(anim: StringName) -> float:
	return float(_attack_weights.get(anim, 1.0))


func _pick_damage() -> Dictionary:
	return {"anim": ANIM_DAMAGE, "frame": animation_frame_damage}


func _after_block_hit() -> void:
	if _block.should_drop():
		unblock()


func _hide_ui_on_death() -> void:
	pass


# ── action lifecycle ──

func _begin_action() -> int:
	_move.kill()
	return _ctx.begin()


# ── damage / death ──

func _apply_damage(amount: float, flash: bool = true) -> void:
	_health.apply(amount)
	damaged.emit(amount)

	if _health.is_dead():
		# Финальный удар: хит-стоп, потом смерть.
		_hit_stop.play(death_hit_stop_duration)
		_die()
		return

	if flash:
		_flash.flash(flash_color, flash_in_duration, flash_out_duration)

	_hit_shake.play(_last_hit_from, hit_shake_preset)
	_hit_stop.play(hit_stop_duration)


func _die() -> void:
	if _state == State.DEAD:
		return

	_state = State.DEAD
	_block.reset()
	_begin_action()
	_hit_shake.kill()
	_flash.kill()
	_juice.kill()

	if animated_sprite:
		animated_sprite.position = _sprite_base_position
		animated_sprite.modulate = Color.WHITE

	_hide_ui_on_death()
	_anim.play(ANIM_DEATH)

	if not death_disappear:
		add_to_group(DeathService.CORPSE_GROUP)

	died.emit()
	_death.play_death()


func _on_blocked_hit(amount: float, from_position: Vector2) -> void:
	var leaked := await _block.on_hit(
		amount, from_position, func() -> void: hit_blocked.emit(amount)
	)
	if _state == State.DEAD:
		return

	if leaked > 0.0:
		_apply_damage(leaked, false)

	_after_block_hit()

var health: float:
	get:
		return _health.health if _health else 0.0
	set(value):
		if _health:

func _play_anim(anim: StringName) -> void:
	_anim.play(anim)


func _kill_tween(tw: Tween) -> void:
	if tw and tw.is_valid():
		tw.kill()


## Публичная «визуальная» реакция блока (без учёта absorbed).
## Используется сабклассами (Enemy reactive block).
func play_block_reaction(amount: float, from_position: Vector2 = Vector2.INF) -> void:
	if is_dead():
		return
	await _block.play_reaction(from_position, func() -> void: hit_blocked.emit(amount))

func init_bar(bar: HealthBar, value: float, max_value: float) -> void:
	if bar:
		bar.setup(value, max_value)


func sync_bar(bar: HealthBar, value: float, max_value: float) -> void:
	if bar:
		bar.update_value(value, max_value)

extends Character
class_name Enemy

@onready var health_bar: HealthBar = $HealthBar

@export var reactive_block_hold: float = 0.35
@export var reactive_block_min_hold: float = 0.15

@export_group("Attack Energy")
## Порог, при котором враг идёт в атаку.
@export var attack_energy_threshold: float = 100.0
## Скорость накопления у «первого стоящего» врага (ед/сек).
@export var energy_rate_front: float = 33.0
## Скорость накопления у остальных (ед/сек).
@export var energy_rate_others: float = 8.0

const ENEMY_GROUP := &"enemies"

var _reactive_block_tween: Tween
var _reactive_pending: bool = false

# ── скрытая энергия атаки ──
var _attack_energy: float = randf_range(50, 65)
var _is_front: bool = false
var _attack_target: Character


func _ready() -> void:
	super()
	init_bar(health_bar, get_health(), get_max_health())
	add_to_group(ENEMY_GROUP)
	if _attack_frames.is_empty():
		_attack_frames = { &"attack": 0 }


# ── энергия атаки ──

func set_attack_target(t: Character) -> void:
	_attack_target = t


func set_is_front(value: bool) -> void:
	_is_front = value


func get_attack_energy() -> float:
	return _attack_energy


func _process(delta: float) -> void:
	if _state == State.DEAD:
		return
	if is_busy():
		return
	if _attack_target == null or not is_instance_valid(_attack_target) or _attack_target.is_dead():
		return

	var rate := energy_rate_front if _is_front else energy_rate_others
	_attack_energy = minf(_attack_energy + rate * delta, attack_energy_threshold)

	if _attack_energy < attack_energy_threshold:
		return

	if _attack_target.is_attacking():
		return
	if _any_enemy_attacking():
		return

	_attack_energy = 0.0
	attack(_attack_target)


func _any_enemy_attacking() -> bool:
	for other in get_tree().get_nodes_in_group(ENEMY_GROUP):
		if other == self or not is_instance_valid(other):
			continue
		if other.is_attacking():
			return true
	return false


# ── урон / хил ──

func heal(amount: float) -> void:
	super.heal(amount)
	sync_bar(health_bar, get_health(), get_max_health())


func take_damage(amount: float, from_position: Vector2 = Vector2.INF) -> void:
	if _state == State.DEAD:
		return

	if _state == State.BLOCKING:
		super.take_damage(amount, from_position)
		return

	if _state == State.IDLE and not _reactive_pending and randf() < character_stats.block_chance:
		_reactive_block(amount, from_position)
		return

	super.take_damage(amount, from_position)


func _reactive_block(amount: float, from_position: Vector2) -> void:
	_reactive_pending = true
	_last_hit_from = from_position

	var leaked := maxf(0.0, amount - get_block_damage_reduction())

	await play_block_reaction(amount, from_position)
	_reactive_pending = false

	if _state == State.DEAD:
		return
	if leaked > 0.0:
		_apply_damage(leaked, false)
	if _state != State.DEAD:
		_schedule_return_to_idle()


func _schedule_return_to_idle() -> void:
	_kill_tween(_reactive_block_tween)
	_reactive_block_tween = create_tween()
	_reactive_block_tween.tween_interval(maxf(reactive_block_hold, reactive_block_min_hold))
	_reactive_block_tween.tween_callback(_return_to_idle_after_block)


func _return_to_idle_after_block() -> void:
	if _state == State.IDLE and animated_sprite and animated_sprite.animation == ANIM_BLOCK:
		_play_anim(ANIM_IDLE)


func _apply_damage(amount: float, flash: bool = true) -> void:
	super(amount, flash)
	sync_bar(health_bar, get_health(), get_max_health())

func _die() -> void:
	super()

func _hide_ui_on_death() -> void:
	if health_bar:
		health_bar.visible = false

const PLAYER_GROUP := &"player"

@export_group("Block Feedback")
## Глобальный freeze-frame при успешном блоке. 0 = выключен.
@export var block_hitstop_duration: float = 0.06

var enemies: Array[Character] = []

var _attack_locked: bool = false
var _attack_turn_active: bool = false
var _block_held: bool = false


func _ready() -> void:
	super()
	add_to_group(PLAYER_GROUP)
	if _attack_frames.is_empty():
		_attack_frames = { &"attack_1": 0, &"attack_2": 1 }
	if _attack_weights.is_empty():
		_attack_weights = { &"attack_1": 1.75, &"attack_2": 1.0 }


func _physics_process(delta: float) -> void:
	_maybe_enter_queued_block()


func _apply_damage(amount: float, flash: bool = true) -> void:
	_emit_screen_shake(amount)
	super(amount, flash)


func _emit_screen_shake(amount: float) -> void:
	var p := HitShakePresets.resolve(hit_shake_preset, _hit_shake.custom)
	var dist := float(p["distance"])
	var max_hp := maxf(get_max_health(), 1.0)
	# 0.5× на «чирик», до 2× на тяжёлый удар по отношению к max HP.
	var scale := clampf(0.5 + (amount / max_hp) * 1.5, 0.5, 2.0)
	screen_shake_requested.emit(dist * scale, float(p["duration"]))


func set_attack_locked(value: bool) -> void:
	var was_locked := _attack_locked
	_attack_locked = value
	if was_locked and not value:
		_maybe_enter_queued_block()


func _unhandled_input(event: InputEvent) -> void:
	if event.is_action_pressed(&"attack"):
		_try_attack()
	elif event.is_action_pressed(&"block"):
		_block_held = true
		_try_block()
	elif event.is_action_released(&"block"):
		_block_held = false
		if _state == State.BLOCKING:
			unblock()


func _try_attack() -> void:
	if _attack_locked or _attack_turn_active or _state == State.DEAD:
		return
	if _state == State.BLOCKING:
		unblock()
	if is_busy():
		return
	if _any_enemy_attacking():
		return

	var enemy := _nearest_enemy()
	if enemy == null:
		return

	_attack_turn_active = true
	attack(enemy)
	_wait_turn_end()


func _try_block() -> void:
	if _attack_locked or _attack_turn_active or _state == State.DEAD:
		return
	if _state == State.BLOCKING or is_busy():
		return
	block()


func _wait_turn_end() -> void:
	if is_busy():
		await turn_finished
	_attack_turn_active = false
	_maybe_enter_queued_block()


func _maybe_enter_queued_block() -> void:
	if not _block_held or _attack_locked or _attack_turn_active:
		return
	if _state != State.IDLE:
		return
	block()


func _nearest_enemy() -> Character:
	var best: Character = null
	var best_d2 := INF
	for e in enemies:
		if not is_instance_valid(e) or e.is_dead():
			continue
		var d2 := global_position.distance_squared_to(e.global_position)
		if d2 < best_d2:
			best_d2 = d2
			best = e
	return best


func _any_enemy_attacking() -> bool:
	for e in enemies:
		if is_instance_valid(e) and e.is_attacking():
			return true
	return false

class_name AnimationHelper
extends RefCounted

const WAIT_TIMEOUT := 3.0

var _host: Node
var _sprite: AnimatedSprite2D


func setup(host: Node, sprite: AnimatedSprite2D) -> void:
	_host = host
	_sprite = sprite


func has(anim: StringName) -> bool:
	return (
		_sprite != null
		and _sprite.sprite_frames != null
		and _sprite.sprite_frames.has_animation(anim)
	)


func play(anim: StringName) -> void:
	if not has(anim):
		return
	if _sprite.animation == anim and _sprite.is_playing():
		return
	_sprite.play(anim)


func frame_time(anim: StringName, frame: int) -> float:
	if not has(anim) or frame <= 0:
		return 0.0
	var speed := _sprite.sprite_frames.get_animation_speed(anim)
	if speed <= 0.0:
		return 0.0
	return float(frame) / speed


func await_end(anim: StringName, ctx: ActionContext, id: int) -> void:
	if not has(anim):
		return
	var elapsed := 0.0
	while ctx.is_current(id) and _sprite.animation == anim and _sprite.is_playing():
		await _host.get_tree().process_frame
		elapsed += _host.get_process_delta_time()
		if elapsed > WAIT_TIMEOUT:
			push_warning("Anim '%s' не завершилась за %.1fs" % [anim, WAIT_TIMEOUT])
			if is_instance_valid(_sprite):
				_sprite.stop()
			return


func await_frame(anim: StringName, frame: int, ctx: ActionContext, id: int) -> void:
	if not has(anim) or frame <= 0:
		return
	var frame_count := _sprite.sprite_frames.get_frame_count(anim)
	var target_frame := clampi(frame, 0, maxi(frame_count - 1, 0))
	var elapsed := 0.0
	while ctx.is_current(id) and _sprite.animation == anim and _sprite.is_playing():
		if _sprite.frame >= target_frame:
			return
		await _sprite.frame_changed
		elapsed += _host.get_process_delta_time()
		if elapsed > WAIT_TIMEOUT:
			push_warning("Anim '%s' не достигла кадра %d за %.1fs" % [anim, target_frame, WAIT_TIMEOUT])
			if is_instance_valid(_sprite):
				_sprite.stop()
			return

func restart(anim: StringName) -> void:
	if not has(anim):
		return
	_sprite.stop()
	_sprite.play(anim)

class_name BlockController
extends RefCounted

const BLOCK_ANIM := &"block"

var _host: Node2D
var _anim: AnimationHelper
var _flash: FlashService
var _hit_shake: HitShakeService
var _juice: CombatJuice
var _get_damage_pick: Callable
var _is_dead: Callable
 
# state
var absorbed: float = 0.0

# config
var max_absorbed: float = 0.0
var damage_reduction: float = 0.0
var lead_ratio: float = 1.0
var flash_color: Color = Color(0.4, 0.65, 1.0, 1.0)
var flash_in: float = 0.03
var flash_out: float = 0.12
var shake_distance: float = 6.0


func setup(
	host: Node2D,
	anim: AnimationHelper,
	flash: FlashService,
	hit_shake: HitShakeService,
	juice: CombatJuice,
	get_damage_pick: Callable,
	is_dead: Callable
) -> void:
	_host = host
	_anim = anim
	_flash = flash
	_hit_shake = hit_shake
	_juice = juice
	_get_damage_pick = get_damage_pick
	_is_dead = is_dead


func reset() -> void:
	absorbed = 0.0


func should_drop() -> bool:
	return max_absorbed > 0.0 and absorbed >= max_absorbed


## Полная реакция: трекинг absorbed + визуал. Возвращает просочившийся урон.
func on_hit(amount: float, from_position: Vector2, on_impact: Callable) -> float:
	var leaked := maxf(0.0, amount - maxf(0.0, damage_reduction))
	absorbed += amount - leaked
	await _play_reaction(from_position, on_impact)
	return leaked


## Только визуал (без трекинга absorbed).
func play_reaction(from_position: Vector2, on_impact: Callable) -> void:
	await _play_reaction(from_position, on_impact)


func _play_reaction(from_position: Vector2, on_impact: Callable) -> void:
	var pick: Dictionary = _get_damage_pick.call()
	var total := _anim.frame_time(pick["anim"], int(pick["frame"]))
	var lead := total * lead_ratio

	if lead > 0.0:
		await _host.get_tree().create_timer(lead).timeout
		if _is_dead.call():
			return

	_anim.play(BLOCK_ANIM)

	var rest := total - lead
	if rest > 0.0:
		await _host.get_tree().create_timer(rest).timeout
		if _is_dead.call():
			return

	on_impact.call()
	_flash.flash(flash_color, flash_in, flash_out)
	_hit_shake.play(from_position, HitShakePresets.Preset.BLOCK_HIT, shake_distance)
	_juice.play_block_recoil(from_position)

class_name CombatJuice
extends RefCounted

var _host: Node2D
var _sprite: AnimatedSprite2D
var _base_pos: Vector2
var _anim: AnimationHelper
var _recoil_tween: Tween

# config
var attack_anticipation_speed: float = 0.45
var attack_impact_speed: float = 2.0
var attack_anticipation_frames: int = 2
var attack_lunge_distance: float = 4.0
var attack_recoil_distance: float = 5.0
var block_recoil_distance: float = 4.0


func setup(host: Node2D, sprite: AnimatedSprite2D, base_pos: Vector2, anim: AnimationHelper) -> void:
	_host = host
	_sprite = sprite
	_base_pos = base_pos
	_anim = anim


func play_lunge(dir: Vector2) -> void:
	if _sprite == null or attack_lunge_distance <= 0.0:
		return
	kill()
	_recoil_tween = _sprite.create_tween()
	_recoil_tween.set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_OUT)
	_recoil_tween.tween_property(
		_sprite, "position", _base_pos - dir * attack_lunge_distance, 0.06
	)
	_recoil_tween.tween_property(_sprite, "position", _base_pos, 0.08)


func play_attack_recoil(dir: Vector2) -> void:
	if _sprite == null or attack_recoil_distance <= 0.0:
		return
	_play_recoil(dir, attack_recoil_distance, 0.05, 0.10)


func play_block_recoil(from_position: Vector2) -> void:
	if _sprite == null or block_recoil_distance <= 0.0:
		return
	var dir := Vector2.UP
	if from_position != Vector2.INF:
		var delta: Vector2 = _host.global_position - from_position
		if delta.length_squared() > 0.0001:
			dir = delta.normalized()
	_play_recoil(dir, block_recoil_distance, 0.05, 0.12)


func _play_recoil(dir: Vector2, distance: float, out_dur: float, in_dur: float) -> void:
	kill()
	_recoil_tween = _sprite.create_tween()
	_recoil_tween.set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	_recoil_tween.tween_property(_sprite, "position", _base_pos + dir * distance, out_dur)
	_recoil_tween.tween_property(_sprite, "position", _base_pos, in_dur)


func play_anticipation(anim: StringName, impact_frame: int, ctx: ActionContext, id: int) -> void:
	if _sprite == null or not _anim.has(anim):
		return
	if attack_anticipation_frames <= 0 or attack_anticipation_speed >= 1.0:
		return

	var frame_count := _sprite.sprite_frames.get_frame_count(anim)
	var target_frame := clampi(
		impact_frame - attack_anticipation_frames, 0, maxi(frame_count - 1, 0)
	)
	var old_speed := _sprite.speed_scale
	_sprite.speed_scale = attack_anticipation_speed

	while ctx.is_current(id) \
			and _sprite.animation == anim \
			and _sprite.is_playing() \
			and _sprite.frame < target_frame:
		await _sprite.frame_changed

	if not is_instance_valid(_sprite):
		return
	_sprite.speed_scale = old_speed


func play_impact(anim: StringName) -> void:
	if _sprite == null or attack_impact_speed <= 1.0:
		return
	var old_speed := _sprite.speed_scale
	_sprite.speed_scale = attack_impact_speed
	var impact_time := _anim.frame_time(anim, 1)
	await _sprite.get_tree().create_timer(maxf(impact_time * 0.5, 0.02)).timeout
	if is_instance_valid(_sprite):
		_sprite.speed_scale = old_speed


func kill() -> void:
	if _recoil_tween and _recoil_tween.is_valid():
		_recoil_tween.kill()

class_name DeathService
extends RefCounted

const CORPSE_GROUP := &"corpses"
const DEATH_ANIM := &"death"

var _host: Node2D
var _sprite: AnimatedSprite2D
var _base_pos: Vector2
var _anim: AnimationHelper
var _get_last_hit_from: Callable

var _corpse_landing_pos: Vector2 = Vector2.INF

# config
var arc_height: float = 8.0
var knockback_distance: float = 120.0
var knockback_speed: float = 500.0
var knockback_angle_jitter: float = 0.2
var knockback_distance_jitter: float = 0.3
var min_corpse_distance: float = 150.0
var landing_attempts: int = 40
var landing_bounds: Rect2 = Rect2()
var landing_viewport_pad: Vector2 = Vector2(60.0, 60.0)
var disappear: bool = false
var corpse_alpha: float = 1.0
var flying_z: int = 2000
var corpse_z: int = -1
var corpse_z_sort_span: int = 1500
var corpse_sort_y_range: Vector2 = Vector2.ZERO
var fade_duration: float = 0.5
var fade_delay: float = 0.0


func setup(
	host: Node2D,
	sprite: AnimatedSprite2D,
	base_pos: Vector2,
	anim: AnimationHelper,
	get_last_hit_from: Callable
) -> void:
	_host = host
	_sprite = sprite
	_base_pos = base_pos
	_anim = anim
	_get_last_hit_from = get_last_hit_from


func get_corpse_anchor() -> Vector2:
	if _corpse_landing_pos != Vector2.INF:
		return _corpse_landing_pos
	return _host.global_position


func play_death() -> void:
	var from: Vector2 = _get_last_hit_from.call()
	if from == Vector2.INF:
		from = _host.global_position + Vector2.DOWN

	var base_dir := _host.global_position - from
	if base_dir.length_squared() < 0.0001:
		base_dir = Vector2.UP
	base_dir = base_dir.normalized()

	var land := _find_corpse_landing(base_dir)
	_corpse_landing_pos = land

	_host.z_index = flying_z

	var dist := _host.global_position.distance_to(land)
	var speed := maxf(knockback_speed, 1.0)
	var fly_duration := maxf(dist / speed, 0.08)

	var knockback := _host.create_tween()
	knockback.set_trans(Tween.TRANS_QUAD).set_ease(Tween.EASE_OUT)
	knockback.tween_property(_host, "global_position", land, fly_duration)

	_play_arc(fly_duration)

	if not disappear:
		_drop_z_when_settled(knockback)

	_start_fade()


func _play_arc(fly_duration: float) -> void:
	if _sprite == null or arc_height <= 0.0:
		return
	var arc := _host.create_tween()
	arc.set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_OUT)
	arc.tween_property(
		_sprite, "position", _base_pos + Vector2(0.0, -arc_height), fly_duration * 0.4
	)
	arc.tween_property(_sprite, "position", _base_pos, fly_duration * 0.6)


func _drop_z_when_settled(knockback_tween: Tween) -> void:
	if knockback_tween and knockback_tween.is_valid() and knockback_tween.is_running():
		await knockback_tween.finished
	await _await_death_last_frame()
	_host.z_index = _compute_corpse_z()


func _await_death_last_frame() -> void:
	if not _anim.has(DEATH_ANIM):
		return
	if _sprite == null or _sprite.animation != DEATH_ANIM:
		return
	var last_frame := maxi(_sprite.sprite_frames.get_frame_count(DEATH_ANIM) - 1, 0)
	if _sprite.frame >= last_frame:
		return
	var elapsed := 0.0
	while _sprite.animation == DEATH_ANIM and _sprite.is_playing():
		if _sprite.frame >= last_frame:
			return
		await _sprite.frame_changed
		elapsed += _host.get_process_delta_time()
		if elapsed > AnimationHelper.WAIT_TIMEOUT:
			return


func _find_corpse_landing(base_dir: Vector2) -> Vector2:
	var corpses := _host.get_tree().get_nodes_in_group(CORPSE_GROUP)
	var bounds := _resolve_landing_bounds()
	var best := _host.global_position + base_dir * knockback_distance
	var best_min_dist := -1.0

	for _attempt in landing_attempts:
		var angle := 0.0
		if knockback_angle_jitter > 0.0:
			angle = randf_range(-knockback_angle_jitter, knockback_angle_jitter)

		var dist := knockback_distance
		if knockback_distance_jitter > 0.0:
			dist *= randf_range(1.0 - knockback_distance_jitter, 1.0 + knockback_distance_jitter)

		var candidate := _host.global_position + base_dir.rotated(angle) * dist
		if bounds.size != Vector2.ZERO:
			candidate = _clamp_to_rect(candidate, bounds)

		var min_dist := _min_dist_to_corpses(candidate, corpses)
		if min_dist >= min_corpse_distance:
			return candidate
		if min_dist > best_min_dist:
			best_min_dist = min_dist
			best = candidate

	return best


func _min_dist_to_corpses(pos: Vector2, corpses: Array) -> float:
	var min_dist := INF
	for corpse in corpses:
		if corpse == _host or not is_instance_valid(corpse):
			continue
		var corpse_pos: Vector2 = corpse.global_position
		if corpse.has_method("get_corpse_anchor"):
			corpse_pos = corpse.get_corpse_anchor()
		min_dist = minf(min_dist, pos.distance_to(corpse_pos))
	return min_dist


func _resolve_landing_bounds() -> Rect2:
	if landing_bounds.size != Vector2.ZERO:
		return landing_bounds
	var viewport := _host.get_viewport()
	if viewport == null:
		return Rect2()
	var visible_rect := viewport.get_visible_rect()
	var pad := landing_viewport_pad
	var inner := Rect2(visible_rect.position + pad, visible_rect.size - pad * 2.0)
	if inner.size.x <= 0.0 or inner.size.y <= 0.0:
		return Rect2()
	return inner


func _clamp_to_rect(pos: Vector2, rect: Rect2) -> Vector2:
	return Vector2(
		clampf(pos.x, rect.position.x, rect.end.x),
		clampf(pos.y, rect.position.y, rect.end.y)
	)


func _compute_corpse_z() -> int:
	if corpse_z_sort_span <= 0:
		return corpse_z
	var y_range := _resolve_corpse_sort_y_range()
	if y_range.y <= y_range.x:
		return corpse_z
	var t := clampf(inverse_lerp(y_range.x, y_range.y, _corpse_landing_pos.y), 0.0, 1.0)
	var z := corpse_z - int(round((1.0 - t) * float(corpse_z_sort_span)))
	return maxi(z, -4096)


func _resolve_corpse_sort_y_range() -> Vector2:
	if corpse_sort_y_range.y > corpse_sort_y_range.x:
		return corpse_sort_y_range
	var bounds := _resolve_landing_bounds()
	if bounds.size.y > 0.0:
		return Vector2(bounds.position.y, bounds.end.y)
	return Vector2.ZERO


func _start_fade() -> void:
	if not disappear and is_equal_approx(corpse_alpha, 1.0):
		return
	var fade := _host.create_tween()
	if fade_delay > 0.0:
		fade.tween_interval(fade_delay)
	if disappear:
		fade.tween_property(_host, "modulate:a", 0.0, maxf(fade_duration, 0.01))
		fade.tween_callback(_host.queue_free)
	elif fade_duration > 0.0:
		fade.tween_property(_host, "modulate:a", corpse_alpha, fade_duration)
	else:
		_host.modulate.a = corpse_alpha

class_name HitShakeService
extends RefCounted

var _sprite: AnimatedSprite2D
var _base_pos: Vector2
var _get_origin: Callable
var _is_invalid: Callable
var _tween: Tween

# config
var distance_jitter: float = 0.0
var duration_jitter: float = 0.0
var decay_jitter: float = 0.0
var custom: Dictionary = {
    "distance": 14.0, "duration": 0.3, "oscillations": 7,
    "decay": 0.65, "return_mult": 1.5
}


func setup(
    sprite: AnimatedSprite2D,
    base_pos: Vector2,
    get_origin: Callable,
    is_invalid: Callable
) -> void:
    _sprite = sprite
    _base_pos = base_pos
    _get_origin = get_origin
    _is_invalid = is_invalid


func play(from_position: Vector2, preset: int, distance_override := -1.0) -> void:
    if _sprite == null or _is_invalid.call():
        return

    var data := HitShakePresets.resolve(preset, custom)
    var distance: float = distance_override if distance_override >= 0.0 else data["distance"]
    var duration: float = data["duration"]
    var oscillations: int = maxi(1, data["oscillations"])
    var decay: float = data["decay"]
    var return_mult: float = data["return_mult"]

    if distance_jitter > 0.0:
        distance *= randf_range(1.0 - distance_jitter, 1.0 + distance_jitter)
    if duration_jitter > 0.0:
        duration *= randf_range(1.0 - duration_jitter, 1.0 + duration_jitter)
    if decay_jitter > 0.0:
        decay = clampf(decay + randf_range(-decay_jitter, decay_jitter), 0.0, 1.0)

    var dir := Vector2.UP
    if from_position != Vector2.INF:
        var delta: Vector2 = _get_origin.call() - from_position
        if delta.length_squared() > 0.0001:
            dir = delta.normalized()

    kill()
    _tween = _sprite.create_tween()
    _tween.set_trans(Tween.TRANS_SINE).set_ease(Tween.EASE_IN_OUT)

    var half_steps := oscillations * 2
    var step_duration := maxf(duration / float(half_steps + 1), 0.01)
    var amp := distance

    for i in range(half_steps):
        var target := _base_pos + dir * amp
        if i % 2 == 1:
            target = _base_pos - dir * amp
        var tweener := _tween.tween_property(_sprite, "position", target, step_duration)
        if i == 0:
            tweener.set_ease(Tween.EASE_OUT)
        amp *= decay

    _tween.tween_property(
        _sprite, "position", _base_pos, maxf(step_duration * return_mult, 0.01)
    ).set_ease(Tween.EASE_OUT)


func kill() -> void:
    if _tween and _tween.is_valid():
        _tween.kill()

class_name HitStopService
extends RefCounted

var _host: Node
var _active: int = 0
var _saved_time_scale: float = 1.0


func setup(host: Node) -> void:
	_host = host


func play(duration: float) -> void:
	if duration <= 0.0 or _host == null:
		return

	_active += 1
	if _active == 1:
		_saved_time_scale = Engine.time_scale
		Engine.time_scale = .1

	await _host.get_tree().create_timer(duration, true, false, true).timeout

	_active -= 1
	if _active == 0:
		Engine.time_scale = _saved_time_scale

class_name MovementService
extends RefCounted

var _host: Node2D
var _ctx: ActionContext
var _tween: Tween


func setup(host: Node2D, ctx: ActionContext) -> void:
    _host = host
    _ctx = ctx


func move_to(
    target: Vector2,
    duration: float,
    id: int,
    trans := Tween.TRANS_QUAD,
    ease_type := Tween.EASE_IN_OUT
) -> bool:
    kill()
    _tween = _host.create_tween()
    _tween.set_trans(trans).set_ease(ease_type)
    _tween.tween_property(_host, "global_position", target, duration)
    return await _wait(id)


func _wait(id: int) -> bool:
    await _host.get_tree().process_frame
    while _ctx.is_current(id) and _tween.is_valid() and _tween.is_running():
        await _host.get_tree().process_frame
    return _ctx.is_current(id)


func kill() -> void:
    if _tween and _tween.is_valid() and _tween.is_running():
        _tween.kill()
